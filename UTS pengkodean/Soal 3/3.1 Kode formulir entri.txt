import React, { useState, useEffect } from 'react';
import { Plus, Trash2, CheckCircle2, XCircle, Save, Clock, User } from 'lucide-react';


export default function App() {
  // State for Journal Entries
  const [entries, setEntries] = useState([
    { id: 'row-1', account: '', debit: '', credit: '' },
    { id: 'row-2', account: '', debit: '', credit: '' },
  ]);


  // State for Audit Logs (Simulated Database Table)
  const [auditLogs, setAuditLogs] = useState([]);
 
  // State for Notifications
  const [notification, setNotification] = useState(null);


  // List of Account Options
  const accountOptions = [
    "Kas Kecil",
    "Kas Bank",
    "Deposito Berjangka (≤ 3 Bulan)",
    "Surat Berharga Lancar",
    "Beban Operasional",
    "Hutang Usaha",
    "Aset Tetap"
  ];


  // Identifikasi akun yang termasuk Kas / Setara Kas (4 item pertama)
  const cashAccounts = accountOptions.slice(0, 4);


  // Derived State for Calculations
  const totalDebit = entries.reduce((acc, curr) => acc + (parseFloat(curr.debit) || 0), 0);
  const totalCredit = entries.reduce((acc, curr) => acc + (parseFloat(curr.credit) || 0), 0);
 
  const totalCashDebit = entries
    .filter(entry => cashAccounts.includes(entry.account))
    .reduce((acc, curr) => acc + (parseFloat(curr.debit) || 0), 0);
   
  const totalCashCredit = entries
    .filter(entry => cashAccounts.includes(entry.account))
    .reduce((acc, curr) => acc + (parseFloat(curr.credit) || 0), 0);


  // 1. Keseimbangan Debit & Kredit (Toleransi selisih < 0.01)
  const discrepancy = Math.abs(totalDebit - totalCredit);
  const hasValues = totalDebit > 0 || totalCredit > 0;
  const isBalanced = discrepancy < 0.01 && hasValues;


  // 2. Seluruh baris memiliki akun yang terisi
  const allAccountsFilled = entries.length > 0 && entries.every(entry => entry.account !== '');


  // 3. Arus Kas Bersih Keluar (Total Kredit Kas > Total Debit Kas)
  const isNetCashOutflow = totalCashCredit > totalCashDebit;


  // 4. Tidak ada baris bernominal nol (Debit atau Kredit harus terisi > 0)
  const noZeroValueRows = entries.length > 0 && entries.every(entry =>
    (parseFloat(entry.debit) || 0) > 0 || (parseFloat(entry.credit) || 0) > 0
  );


  // Status kelayakan tombol submit
  const isReadyToSubmit = isBalanced && allAccountsFilled && isNetCashOutflow && noZeroValueRows;


  // Format currency
  const formatCurrency = (amount) => {
    return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR' }).format(amount || 0);
  };


  // Generate unique ID for new rows
  const generateId = () => Math.random().toString(36).substr(2, 9);


  // Handle adding a new row
  const handleAddRow = () => {
    setEntries([...entries, { id: generateId(), account: '', debit: '', credit: '' }]);
  };


  // Handle removing a row
  const handleRemoveRow = (idToRemove) => {
    if (entries.length <= 2) {
      alert("Minimal harus ada 2 baris entri jurnal.");
      return;
    }
    setEntries(entries.filter((entry) => entry.id !== idToRemove));
  };


  // Handle input changes
  const handleInputChange = (id, field, value) => {
    const updatedEntries = entries.map((entry) => {
      if (entry.id === id) {
        let newEntry = { ...entry, [field]: value };
        // Clear opposite field automatically
        if (field === 'debit' && parseFloat(value) > 0) newEntry.credit = '';
        if (field === 'credit' && parseFloat(value) > 0) newEntry.debit = '';
        return newEntry;
      }
      return entry;
    });
    setEntries(updatedEntries);
  };


  // Handle Form Submission
  const handleSubmit = () => {
    if (!isReadyToSubmit) return;


    const currentTimestamp = new Date().toISOString();
    const mockUserId = "USR-" + Math.floor(Math.random() * 9000 + 1000);
   
    const newLog = {
      id: generateId(),
      timestamp: currentTimestamp,
      userId: mockUserId,
      action: "INSERT_VOUCHER_KAS_KELUAR",
      value: totalDebit,
      details: entries
    };


    setAuditLogs([newLog, ...auditLogs]);
   
    setNotification(`Sukses! Voucher Kas Keluar sebesar ${formatCurrency(totalDebit)} berhasil disimpan.`);
   
    // Reset Form
    setEntries([
      { id: generateId(), account: '', debit: '', credit: '' },
      { id: generateId(), account: '', debit: '', credit: '' },
    ]);


    setTimeout(() => {
      setNotification(null);
    }, 5000);
  };


  return (
    <div className="min-h-screen bg-slate-50 p-4 md:p-8 font-sans text-slate-800">
      <div className="max-w-5xl mx-auto space-y-6">
       
        {/* Header */}
        <header className="bg-white p-6 rounded-xl shadow-sm border border-slate-200">
          <h1 className="text-2xl font-bold text-slate-800 flex items-center gap-2">
            Modul Voucher Entri Transaksi Kas Keluar
          </h1>
          <p className="text-slate-500 mt-1">Sistem pencatatan jurnal pengeluaran kas dengan validasi ketat sesuai standar akuntansi.</p>
        </header>


        {/* Notification Alert */}
        {notification && (
          <div className="bg-emerald-50 border border-emerald-200 text-emerald-800 px-4 py-3 rounded-lg flex items-center gap-3">
            <CheckCircle2 className="h-5 w-5 text-emerald-600" />
            <span className="font-medium">{notification}</span>
          </div>
        )}


        {/* Main Form Area */}
        <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
          <div className="p-6 border-b border-slate-100">
            <h2 className="text-lg font-semibold mb-4">Detail Jurnal</h2>
           
            <div className="overflow-x-auto">
              <table className="w-full text-left border-collapse min-w-[600px]">
                <thead>
                  <tr className="border-b-2 border-slate-200">
                    <th className="pb-3 pt-2 px-2 text-sm font-semibold text-slate-600 w-12">No</th>
                    <th className="pb-3 pt-2 px-2 text-sm font-semibold text-slate-600 w-1/3">Nama Akun</th>
                    <th className="pb-3 pt-2 px-2 text-sm font-semibold text-slate-600 w-1/4">Debit (Rp)</th>
                    <th className="pb-3 pt-2 px-2 text-sm font-semibold text-slate-600 w-1/4">Kredit (Rp)</th>
                    <th className="pb-3 pt-2 px-2 text-sm font-semibold text-slate-600 w-16 text-center">Aksi</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100">
                  {entries.map((entry, index) => (
                    <tr key={entry.id} className="hover:bg-slate-50/50 transition-colors">
                      <td className="py-3 px-2 text-sm text-slate-500">{index + 1}</td>
                      <td className="py-3 px-2">
                        <select
                          value={entry.account}
                          onChange={(e) => handleInputChange(entry.id, 'account', e.target.value)}
                          className="w-full bg-white border border-slate-300 rounded-md px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/50"
                        >
                          <option value="" disabled>Pilih Akun...</option>
                          {accountOptions.map(acc => (
                            <option key={acc} value={acc}>{acc}</option>
                          ))}
                        </select>
                      </td>
                      <td className="py-3 px-2">
                        <input
                          type="number"
                          min="0"
                          placeholder="0"
                          value={entry.debit}
                          onChange={(e) => handleInputChange(entry.id, 'debit', e.target.value)}
                          disabled={parseFloat(entry.credit) > 0}
                          className="w-full border border-slate-300 rounded-md px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/50 disabled:bg-slate-100 disabled:text-slate-400"
                        />
                      </td>
                      <td className="py-3 px-2">
                        <input
                          type="number"
                          min="0"
                          placeholder="0"
                          value={entry.credit}
                          onChange={(e) => handleInputChange(entry.id, 'credit', e.target.value)}
                          disabled={parseFloat(entry.debit) > 0}
                          className="w-full border border-slate-300 rounded-md px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/50 disabled:bg-slate-100 disabled:text-slate-400"
                        />
                      </td>
                      <td className="py-3 px-2 text-center">
                        <button
                          onClick={() => handleRemoveRow(entry.id)}
                          disabled={entries.length <= 2}
                          className="p-2 text-slate-400 hover:text-red-500 hover:bg-red-50 rounded-md disabled:opacity-50 disabled:hover:bg-transparent transition-colors"
                          title="Hapus Baris"
                        >
                          <Trash2 className="h-4 w-4" />
                        </button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>


            <button
              onClick={handleAddRow}
              className="mt-4 flex items-center gap-2 text-sm text-blue-600 hover:text-blue-700 font-medium px-3 py-2 rounded-md hover:bg-blue-50 transition-colors"
            >
              <Plus className="h-4 w-4" /> Tambah Baris
            </button>
          </div>


          {/* Summary & Validation Section */}
          <div className="p-6 bg-slate-50 flex flex-col lg:flex-row items-start justify-between gap-6">
           
            {/* Total Counters */}
            <div className="w-full lg:w-1/3 grid grid-cols-2 lg:grid-cols-1 xl:grid-cols-2 gap-4">
              <div className="bg-white p-4 rounded-lg border border-slate-200 shadow-sm">
                <p className="text-sm text-slate-500 mb-1">Total Debit</p>
                <p className="text-xl font-bold text-slate-800">{formatCurrency(totalDebit)}</p>
              </div>
              <div className="bg-white p-4 rounded-lg border border-slate-200 shadow-sm">
                <p className="text-sm text-slate-500 mb-1">Total Kredit</p>
                <p className="text-xl font-bold text-slate-800">{formatCurrency(totalCredit)}</p>
              </div>
            </div>


            {/* Validation Matrix */}
            <div className="w-full lg:w-1/2 bg-white p-4 rounded-lg border border-slate-200 shadow-sm">
              <h3 className="text-sm font-semibold text-slate-700 mb-3">Persyaratan Sistem:</h3>
              <ul className="text-sm space-y-2.5">
                <li className={`flex items-center gap-2 ${isBalanced ? 'text-emerald-600' : 'text-rose-500'}`}>
                  {isBalanced ? <CheckCircle2 className="h-4 w-4" /> : <XCircle className="h-4 w-4" />}
                  Keseimbangan Debit & Kredit {discrepancy >= 0.01 && `(Selisih: ${formatCurrency(discrepancy)})`}
                </li>
                <li className={`flex items-center gap-2 ${allAccountsFilled ? 'text-emerald-600' : 'text-rose-500'}`}>
                  {allAccountsFilled ? <CheckCircle2 className="h-4 w-4" /> : <XCircle className="h-4 w-4" />}
                  Seluruh baris memiliki akun terisi
                </li>
                <li className={`flex items-center gap-2 ${isNetCashOutflow ? 'text-emerald-600' : 'text-rose-500'}`}>
                  {isNetCashOutflow ? <CheckCircle2 className="h-4 w-4" /> : <XCircle className="h-4 w-4" />}
                  Arus Kas Bersih Keluar (Kredit Kas &gt; Debit Kas)
                </li>
                <li className={`flex items-center gap-2 ${noZeroValueRows ? 'text-emerald-600' : 'text-rose-500'}`}>
                  {noZeroValueRows ? <CheckCircle2 className="h-4 w-4" /> : <XCircle className="h-4 w-4" />}
                  Tidak ada baris bernominal nol
                </li>
              </ul>
            </div>


            {/* Submit Button */}
            <div className="w-full lg:w-auto flex flex-col items-end">
              <button
                onClick={handleSubmit}
                disabled={!isReadyToSubmit}
                className="flex items-center justify-center gap-2 w-full lg:w-auto px-6 py-4 rounded-lg font-semibold text-white transition-all disabled:opacity-50 disabled:cursor-not-allowed bg-blue-600 hover:bg-blue-700 shadow-md hover:shadow-lg disabled:shadow-none"
              >
                <Save className="h-5 w-5" />
                Submit Jurnal
              </button>
            </div>
          </div>
        </div>


        {/* Audit Log Section */}
        <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
          <div className="p-6 border-b border-slate-100 bg-slate-800 text-white flex justify-between items-center">
            <h2 className="text-lg font-semibold flex items-center gap-2">
              <Clock className="h-5 w-5 text-slate-300" />
              Riwayat Audit
            </h2>
            <span className="text-xs bg-slate-700 px-2 py-1 rounded text-slate-300 font-mono">
              [Simulasi Database]
            </span>
          </div>
         
          <div className="p-0 overflow-x-auto">
            {auditLogs.length === 0 ? (
              <div className="p-8 text-center text-slate-500 text-sm">
                Belum ada transaksi tersimpan pada sesi ini.
              </div>
            ) : (
              <table className="w-full text-left text-sm border-collapse min-w-[600px]">
                <thead className="bg-slate-50 text-slate-600 border-b border-slate-200">
                  <tr>
                    <th className="py-3 px-4 font-semibold w-1/4">Timestamp</th>
                    <th className="py-3 px-4 font-semibold w-1/4">User</th>
                    <th className="py-3 px-4 font-semibold w-1/4">Action</th>
                    <th className="py-3 px-4 font-semibold w-1/4 text-right">Nilai Total</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100 font-mono">
                  {auditLogs.map((log) => (
                    <tr key={log.id} className="hover:bg-slate-50 transition-colors">
                      <td className="py-3 px-4 text-slate-500 text-xs">
                        {new Date(log.timestamp).toLocaleString('id-ID')}
                      </td>
                      <td className="py-3 px-4">
                        <span className="flex items-center gap-1.5 bg-slate-100 text-slate-700 px-2 py-1 rounded w-fit text-xs border border-slate-200">
                          <User className="h-3 w-3" /> {log.userId}
                        </span>
                      </td>
                      <td className="py-3 px-4">
                        <span className="bg-blue-50 text-blue-700 px-2 py-1 rounded text-xs border border-blue-100 font-semibold">
                          {log.action}
                        </span>
                      </td>
                      <td className="py-3 px-4 text-right font-semibold text-slate-700">
                        {formatCurrency(log.value)}
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            )}
          </div>
        </div>


      </div>
    </div>
  );
}
